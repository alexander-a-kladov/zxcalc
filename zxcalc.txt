    DEVICE ZXSPECTRUM128

    ORG #8000           ; Программа будет находиться по адресу 32768

Start:
    ; --- Настройка параметров для функции сложения ---
    LD HL, Num1         ; HL = Адрес Числа 1
    LD B, Num1_Len      ; B  = Длина Числа 1 (2 байта)
    
    LD DE, Num2         ; DE = Адрес Числа 2
    LD C, Num2_Len      ; C  = Длина Числа 2 (4 байта)
    
    LD IX, Result       ; IX = Буфер для результата (размер 5 байт)

    ; --- Вызов функции сложения ---
    CALL AddVariable

    LD HL, Result
    LD A, (ResLen)
    LD B,A
    CALL PrintBigHex

    RET                 ; Возврат в BASIC

; =====================================================================
; Окончательно исправленная функция сложения (БЕЗ потери флага переноса)
; =====================================================================
AddVariable:
    AND A               ; Единственный сброс Carry перед стартом (C = 0)
    LD IY,ResLen
    LD (IY+0),A
LoopBothAdd:
    ; Безопасная проверка: закончилось ли Число 1 (B)?
    LD A, B
    INC A               ; Временно прибавляем 1
    DEC A               ; И вычитаем 1. Это меняет только флаг Z! Carry НЕ ТРОГАЕТСЯ.
    JR Z, LongTwoAdd    ; Если B=0, переходим к обработке остатка Числа 2

    ; Безопасная проверка: закончилось ли Число 2 (C)?
    LD A, C
    INC A : DEC A       ; Проверка регистра C без изменения Carry
    JR Z, LongOneAdd    ; Если C=0, переходим к обработке остатка Числа 1

    ; --- Этап 1: Есть байты в обоих числах ---
    LD A, (DE)          
    ADC A, (HL)         ; Побайтовое сложение с учетом Carry
    LD (IX+0), A        

    INC HL : INC DE : INC IX : INC (IY+0)
    DEC B : DEC C       
    JR LoopBothAdd      ; Повторяем цикл для общих байт

; --- Этап 2a: Число 1 закончилось, обрабатываем остаток Числа 2 ---
LongTwoAdd:
    ; Проверяем, есть ли вообще байты в Числе 2 перед входом в цикл
    LD A, C : INC A : DEC A
    JR Z, FinalCarryAdd ; Если C=0, то складывать больше нечего

LongTwoLoop:
    LD A, (DE)          
    ADC A, 0            ; Складываем байт Числа 2 + ТЕКУЩИЙ ЖИВОЙ CARRY
    LD (IX+0), A
    
    INC DE : INC IX : INC (IY+0)
    DEC C               ; DEC C меняет только флаг Z, флаг Carry остается целым!
    JR NZ, LongTwoLoop  ; Крутим цикл, пока C не станет 0
    JR FinalCarryAdd

; --- Этап 2b: Число 2 закончилось, обрабатываем остаток Числа 1 ---
LongOneAdd:
    LD A, B : INC A : DEC A
    JR Z, FinalCarryAdd

LongOneLoop:
    LD A, (HL)          
    ADC A, 0            ; Складываем байт Числа 1 + ТЕКУЩИЙ ЖИВОЙ CARRY
    LD (IX+0), A
    
    INC HL : INC IX : INC (IY+0)
    DEC B               ; DEC B меняет только флаг Z, флаг Carry остается целым!
    JR NZ, LongOneLoop  ; Крутим цикл, пока B не станет 0

; --- Финал: Запись последнего переноса ---
FinalCarryAdd:
    LD A, 0
    ADC A, 0            ; Вытаскиваем самый последний перенос переполнения (0 или 1)
    LD (IX+0), A        
    RET

; =====================================================================
; АРХИТЕКТУРНО ЧИСТАЯ ФУНКЦИЯ: Печать длинного числа в HEX формате
; Вход: HL = Адрес начала числа (младший байт), B = Длина в байтах
; =====================================================================
PrintBigHex:
    ; Сдвигаем HL на самый старший байт: HL = HL + B - 1
    ; Проходим по массиву от старших адресов к младшим, НЕ трогая регистр IY.
    PUSH HL
    LD E, B
    LD D, 0
    ADD HL, DE          
    DEC HL              ; HL теперь указывает точно на старший байт числа
    EX (SP), HL         ; Сохраняем этот старший адрес в стек, возвращаем начальный HL
    POP DE              ; Теперь DE = Адрес самого старшего байта числа

PrintBigLoop:
    LD A, (DE)          ; Читаем байт из памяти (через безопасный DE)
    
    PUSH BC             ; Сохраняем счетчик байт (B)
    PUSH DE             ; Сохраняем текущий указатель адреса (DE)
    CALL PrintHexByte   ; Печатаем байт гарантированно безопасно
    
    ; Выводим пробел между байтами для читаемости
    LD A, " "           
    RST #10             
    
    POP DE              ; Восстанавливаем указатель адреса
    POP BC              ; Восстанавливаем счетчик байт
    
    DEC DE              ; Двигаемся к более младшему байту памяти
    DJNZ PrintBigLoop   ; Повторяем цикл, пока B не станет 0
    RET

; =====================================================================
; Функция: Безопасная печать одного байта из А (выводит строго символы 0-F)
; =====================================================================
PrintHexByte:
    PUSH AF             ; Сохраняем оригинальный байт
    ; 1. Выводим старшие 4 бита
    RRCA : RRCA : RRCA : RRCA 
    CALL PrintNibble
    ; 2. Восстанавливаем оригинальный байт
    POP AF              
    ; (Проваливаемся в PrintNibble для вывода младших 4 бит)

PrintNibble:
    AND #0F             ; Маска оставляет только нижние 4 бита (число строго 0..15)
    CP 10
    JR C, IsDigit
    ADD A, 7            ; Коррекция для букв A-F
IsDigit:
    ADD A, "0"          ; Перевод полубайта в ASCII-код символа
    RST #10             ; Вызов ПЗУ для печати легального символа
    RET


; =====================================================================
; Данные (Тестовый пример)
; Числа хранятся в формате Little Endian (младший байт первый)
; =====================================================================

; Тест: Складываем 16-битное число и 32-битное число
; Число 1: 0x12EF (В памяти: EF, 12)
Num1:       DB #EF, #12, #FF, #AB
Num1_Len    EQU $ - Num1

; Число 2: 0x00FFED12 (В памяти: 12, ED, FF, 00)
Num2:       DB #20, #30, #50, #12, #ED, #FF, #00, #23, #01
Num2_Len    EQU $ - Num2

; Ожидаемый результат математически: 
;   0x00FFED12 + 0x12EF = 0x01000001
; На экране должно напечататься: C:00 R:01000001

; Выделяем буфер под результат (Максимальная длина чисел + 1 байт на перенос)
Result:     DS 256, 0
ResLen:	    DB	0

    ; Опция 2: Создаем готовый снимок памяти эмулятора (.SNA)
    ; Файл "test.sna" можно просто перетащить в эмулятор (Unreal, Fuse, Spectaculator)
    ; Эмулятор запустится сразу с адреса Start.
    SAVESNA "zxcalc.sna", Start
